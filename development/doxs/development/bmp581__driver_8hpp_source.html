55 DriverResult<void> Init() noexcept {
56 chip_id_ = 0;
57 if (!bus_.EnsureInitialized()) {
58 return DriverResult<void>::failure(DriverError::NotInitialized);
59 }
60 uint8_t id = 0;
61 if (!bus_.ReadRegs(reg::kChipId, &id, 1)) {
62 return DriverResult<void>::failure(DriverError::Bus);
63 }
64 if (id == 0) {
65 return DriverResult<void>::failure(DriverError::InvalidChipId);
66 }
67 DriverError nvm = NvmReadyCheck();
68 if (nvm == DriverError::NvmNotReady) {
69 /* Seen on hardware: a part that answers its chip id with STATUS == 0
70 * (core/NVM boot never completed after power-up). A soft reset
71 * re-runs the NVM boot; give it t_soft_reset + margin and re-check. */
72 if (WriteReg(reg::kCmd, cmd::kSoftReset)) {
73 bus_.DelayUs(timing::kSoftResetUs + timing::kStartupUs);
74 nvm = NvmReadyCheck();
75 }
76 }
77 if (nvm != DriverError::None) {
78 return DriverResult<void>::failure(nvm);
79 }
80 if (id != chip_id::kPrimary && id != chip_id::kSecondary) {
81 return DriverResult<void>::failure(DriverError::InvalidChipId);
82 }
83 chip_id_ = id;
84 return DriverResult<void>::success();
85 }

88 DriverResult<void> SoftReset() noexcept {
89 if (!WriteReg(reg::kCmd, cmd::kSoftReset)) {
90 return DriverResult<void>::failure(DriverError::Bus);
91 }
92 bus_.DelayUs(timing::kSoftResetUs);
93 uint8_t st = 0;
94 if (!bus_.ReadRegs(reg::kIntStatus, &st, 1)) {
95 return DriverResult<void>::failure(DriverError::Bus);
96 }
97 if ((st & int_status::kPorSoftResetComplete) == 0U) {
98 return DriverResult<void>::failure(DriverError::PorSoftReset);
99 }
100 return DriverResult<void>::success();

116 DriverResult<PowerMode> GetPowerMode() noexcept {
117 uint8_t odr = 0;
118 if (!bus_.ReadRegs(reg::kOdrConfig, &odr, 1)) {
119 return DriverResult<PowerMode>::failure(DriverError::Bus);
120 }
121 const uint8_t mode = odr & odr_config::kPowerModeMask;
122 if (mode == static_cast<uint8_t>(PowerMode::Standby)) {
123 const bool deep_disabled = (odr & odr_config::kDeepDisableMask) != 0U;
124 return DriverResult<PowerMode>::success(deep_disabled ? PowerMode::Standby
125 : DeepStandbyOrStandby());
126 }
127 return DriverResult<PowerMode>::success(static_cast<PowerMode>(mode));
128 }

134 DriverResult<void> SetPowerMode(PowerMode mode) noexcept {
135 const auto cur = GetPowerMode();
136 if (!cur.ok()) {
137 return DriverResult<void>::failure(cur.error);
138 }
139 if (cur.value != PowerMode::Standby) {
140 if (!WritePowerModeBits(PowerMode::Standby, /*deep_disable=*/true)) {
141 return DriverResult<void>::failure(DriverError::Bus);
142 }
143 bus_.DelayUs(timing::kStandbyUs);
144 }
145 switch (mode) {
146 case PowerMode::Standby:
147 return DriverResult<void>::success();
148 case PowerMode::DeepStandby:
149 return WritePowerModeBits(PowerMode::Standby, /*deep_disable=*/false)
150 ? DriverResult<void>::success()
151 : DriverResult<void>::failure(DriverError::Bus);
152 case PowerMode::Normal:
153 case PowerMode::Forced:
154 case PowerMode::Continuous:
155 return WritePowerModeBits(mode, /*deep_disable=*/true)
156 ? DriverResult<void>::success()
157 : DriverResult<void>::failure(DriverError::Bus);
158 }
159 return DriverResult<void>::failure(DriverError::InvalidPowerMode);
160 }

164 DriverResult<OsrOdrConfig> GetOsrOdr() noexcept {
165 uint8_t r[2]{};
166 if (!bus_.ReadRegs(reg::kOsrConfig, r, 2)) {
167 return DriverResult<OsrOdrConfig>::failure(DriverError::Bus);
168 }
169 OsrOdrConfig c{};
170 c.osr_temperature = static_cast<Oversampling>(r[0] & osr_config::kTempOsMask);
171 c.osr_pressure = static_cast<Oversampling>((r[0] & osr_config::kPressOsMask) >>
172 osr_config::kPressOsShift);
173 c.pressure_enable = (r[0] & osr_config::kPressEnMask) != 0U;
174 c.odr = static_cast<Odr>((r[1] & odr_config::kOdrMask) >> odr_config::kOdrShift);
175 return DriverResult<OsrOdrConfig>::success(c);

183 DriverResult<void> SetOsrOdr(const OsrOdrConfig& c) noexcept {
184 if (static_cast<uint8_t>(c.odr) < static_cast<uint8_t>(Odr::Hz5)) {
185 const auto cur = GetPowerMode();
186 if (!cur.ok()) {
187 return DriverResult<void>::failure(cur.error);
188 }
189 if (cur.value == PowerMode::DeepStandby) {
190 if (!WritePowerModeBits(PowerMode::Standby, true)) {
191 return DriverResult<void>::failure(DriverError::Bus);
192 }
193 }
194 }
195 uint8_t r[2]{};
196 if (!bus_.ReadRegs(reg::kOsrConfig, r, 2)) {
197 return DriverResult<void>::failure(DriverError::Bus);
198 }
199 r[0] = static_cast<uint8_t>((r[0] & ~(osr_config::kTempOsMask | osr_config::kPressOsMask |
200 osr_config::kPressEnMask)) |
201 (static_cast<uint8_t>(c.osr_temperature) & osr_config::kTempOsMask) |
202 ((static_cast<uint8_t>(c.osr_pressure) << osr_config::kPressOsShift) &
203 osr_config::kPressOsMask) |
204 (c.pressure_enable ? osr_config::kPressEnMask : 0U));
205 r[1] = static_cast<uint8_t>((r[1] & ~odr_config::kOdrMask) |
206 ((static_cast<uint8_t>(c.odr) << odr_config::kOdrShift) &
207 odr_config::kOdrMask));
208 return bus_.WriteRegs(reg::kOsrConfig, r, 2) ? DriverResult<void>::success()
209 : DriverResult<void>::failure(DriverError::Bus);
210 }

239 DriverResult<void> SetIir(const IirConfig& c) noexcept {
240 const auto cur = GetPowerMode();
241 if (!cur.ok()) {
242 return DriverResult<void>::failure(cur.error);
243 }
244 if (cur.value != PowerMode::Standby) {
245 const auto r = SetPowerMode(PowerMode::Standby);
246 if (!r.ok()) return r;
247 }
248 uint8_t r[2]{};
249 if (!bus_.ReadRegs(reg::kDspConfig, r, 2)) {
250 return DriverResult<void>::failure(DriverError::Bus);
251 }
252 r[0] = static_cast<uint8_t>((r[0] & ~(dsp_config::kShadowIirTempMask | dsp_config::kShadowIirPressMask |
253 dsp_config::kIirFlushForcedEnMask)) |
254 (c.shadow_iir_temperature ? dsp_config::kShadowIirTempMask : 0U) |
255 (c.shadow_iir_pressure ? dsp_config::kShadowIirPressMask : 0U) |
256 (c.flush_forced_enable ? dsp_config::kIirFlushForcedEnMask : 0U));
257 r[1] = static_cast<uint8_t>((r[1] & ~(dsp_iir::kTempMask | dsp_iir::kPressMask)) |
258 (static_cast<uint8_t>(c.temperature) & dsp_iir::kTempMask) |
259 ((static_cast<uint8_t>(c.pressure) << dsp_iir::kPressShift) &
260 dsp_iir::kPressMask));
261 if (!bus_.WriteRegs(reg::kDspConfig, r, 2)) {
262 return DriverResult<void>::failure(DriverError::Bus);
263 }
264 if (cur.value != PowerMode::Standby && cur.value != PowerMode::DeepStandby) {
265 return SetPowerMode(cur.value);
266 }
267 return DriverResult<void>::success();
268 }

273 DriverResult<void> ConfigureInterrupt(const InterruptConfig& c) noexcept {
274 uint8_t cfg = 0;
275 if (!bus_.ReadRegs(reg::kIntConfig, &cfg, 1)) {
276 return DriverResult<void>::failure(DriverError::Bus);
277 }
278 if (!WriteReg(reg::kIntSource, 0x00)) {
279 return DriverResult<void>::failure(DriverError::Bus);
280 }
281 uint8_t st = 0;
282 (void)bus_.ReadRegs(reg::kIntStatus, &st, 1);
283 cfg = static_cast<uint8_t>((cfg & ~(int_config::kModeMask | int_config::kPolMask | int_config::kOdMask |
284 int_config::kEnableMask)) |
285 (c.mode == IntMode::Latched ? int_config::kModeMask : 0U) |
286 (c.polarity == IntPolarity::ActiveHigh ? int_config::kPolMask : 0U) |
287 (c.drive == IntDrive::OpenDrain ? int_config::kOdMask : 0U) |
288 (c.enable ? int_config::kEnableMask : 0U));
289 return WriteReg(reg::kIntConfig, cfg) ? DriverResult<void>::success()
290 : DriverResult<void>::failure(DriverError::Bus);
291 }

332 static Sample DecodeSample(const uint8_t d[6], bool pressure_enabled) noexcept {
333 Sample s{};
334 uint32_t t = (static_cast<uint32_t>(d[2]) << 16) | (static_cast<uint32_t>(d[1]) << 8) | d[0];
335 if ((t & 0x800000U) != 0U) {
336 t |= 0xFF000000U;
337 }
338 s.temperature_raw = static_cast<int32_t>(t);
339 s.temperature_c = static_cast<float>(s.temperature_raw) / kTemperatureLsbPerC;
340 if (pressure_enabled) {
341 s.pressure_raw = (static_cast<uint32_t>(d[5]) << 16) | (static_cast<uint32_t>(d[4]) << 8) | d[3];
342 s.pressure_pa = static_cast<float>(s.pressure_raw) / kPressureLsbPerPa;
343 }
344 return s;
345 }

353 DriverResult<uint16_t> ReadNvm(uint8_t nvm_addr) noexcept {
354 if (nvm_addr < nvm::kStartAddr || nvm_addr > nvm::kEndAddr) {
355 return DriverResult<uint16_t>::failure(DriverError::NvmInvalidAddr);
356 }
357 const auto cur = GetPowerMode();
358 if (!cur.ok()) {
359 return DriverResult<uint16_t>::failure(cur.error);
360 }
361 if (cur.value != PowerMode::Standby) {
362 const auto r = SetPowerMode(PowerMode::Standby);
363 if (!r.ok()) return DriverResult<uint16_t>::failure(r.error);
364 }
365 DriverError err = NvmReadyCheck();
366 if (err != DriverError::None) {
367 return DriverResult<uint16_t>::failure(err);
368 }
369 if (!WriteReg(reg::kNvmAddr, static_cast<uint8_t>(nvm_addr & 0x3FU)) ||
370 !WriteReg(reg::kCmd, cmd::kNvmFirst) || !WriteReg(reg::kCmd, cmd::kNvmReadEnable)) {
371 return DriverResult<uint16_t>::failure(DriverError::Bus);
372 }
373 bus_.DelayUs(timing::kNvmReadyReadUs);
374 err = NvmReadyCheck();
375 if (err != DriverError::None) {
376 return DriverResult<uint16_t>::failure(err);
377 }
378 uint8_t d[2]{};
379 if (!bus_.ReadRegs(reg::kNvmDataLsb, d, 2)) {
380 return DriverResult<uint16_t>::failure(DriverError::Bus);
381 }
382 if (cur.value != PowerMode::Standby) {
383 (void)SetPowerMode(cur.value);
384 }
385 return DriverResult<uint16_t>::success(static_cast<uint16_t>((d[1] << 8) | d[0]));
386 }

406 bool WritePowerModeBits(PowerMode mode, bool deep_disable) noexcept {
407 uint8_t odr = 0;
408 if (!bus_.ReadRegs(reg::kOdrConfig, &odr, 1)) {
409 return false;
410 }
411 odr = static_cast<uint8_t>((odr & ~(odr_config::kPowerModeMask | odr_config::kDeepDisableMask)) |
412 (static_cast<uint8_t>(mode) & odr_config::kPowerModeMask) |
413 (deep_disable ? odr_config::kDeepDisableMask : 0U));
414 return WriteReg(reg::kOdrConfig, odr);
415 }

418 PowerMode DeepStandbyOrStandby() noexcept {
419 uint8_t fifo_sel = 0;
420 (void)bus_.ReadRegs(reg::kFifoSel, &fifo_sel, 1);
421 const auto osr = GetOsrOdr();
422 const auto iir = GetIir();
423 if (osr.ok() && iir.ok() && (fifo_sel & 0x03U) == 0U &&
424 static_cast<uint8_t>(osr.value.odr) > static_cast<uint8_t>(Odr::Hz5) &&
425 iir.value.temperature == IirFilter::Bypass && iir.value.pressure == IirFilter::Bypass) {
426 return PowerMode::DeepStandby;
427 }
428 return PowerMode::Standby;
429 }
430
431 DriverError NvmReadyCheck() noexcept {
432 uint8_t st = 0;
433 if (!bus_.ReadRegs(reg::kStatus, &st, 1)) {
434 return DriverError::Bus;
435 }
436 if ((st & status::kNvmCmdErr) != 0U) {
437 return DriverError::NvmCommandError;
438 }
439 if ((st & status::kNvmRdy) == 0U || (st & status::kNvmErr) != 0U) {
440 return DriverError::NvmNotReady;
441 }
442 return DriverError::None;
443 }
